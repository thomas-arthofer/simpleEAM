export const syncDiagramOnOpen = async (
  apolloClient: any,
  diagramData: any,
  sovereigntyOptions?: { enabled: boolean; companyId: string | null }
): Promise<any> => {
  if (!diagramData.elements || !Array.isArray(diagramData.elements)) {
    return diagramData
  }

  debugElementStructure(diagramData.elements)

  let updatedElements = await syncDiagramOnOpenSimple(apolloClient, diagramData.elements)

    ...diagramData,

  }
}

**Analog 2 — find-companion-element-by-id convention:** `findLinkedTextElement` (`textContainerUtils.ts`, lines ~308–334)
```typescript
export const findLinkedTextElement = (
  containerElement: ExcalidrawElement,
  allElements: ExcalidrawElement[]
): ExcalidrawElement | null => {
  if (!containerElement.customData?.isFromDatabase) return null

  // Strategie 1: Direct boundElements Verknüpfung
  if (containerElement.boundElements) {
    const textBound = containerElement.boundElements.find(bound => bound.type === 'text')
    if (textBound) {
      const textElement = allElements.find(el => el.id === textBound.id)
      if (textElement) return textElement
    }
  }
  // Strategie 2: Text-Element mit containerId
  const textWithContainer = allElements.find(
    el => el.type === 'text' && el.containerId === containerElement.id
  )
  if (textWithContainer) return textWithContainer

  // Strategie 3: Text-Element mit mainElementId Verweis
  const textWithMainRef = allElements.find(
    el => el.type === 'text' && el.customData?.mainElementId === containerElement.id
  )
  if (textWithMainRef) return textWithMainRef
  // ...
}
```
This "search by `customData.mainElementId === containerElement.id`" strategy is the **exact lookup key** already used by marker ellipses (`fillEllipse.customData.mainElementId = mainElement.id`, see below) — D-04's marker lookup should use the same `elements.filter(el => el.customData?.mainElementId === mainElement.id && (el.customData?.sovereigntyMarker === 'fill' || el.customData?.sovereigntyMarker === 'ring'))` shape rather than inventing a new lookup mechanism.

### `client/src/components/diagrams/components/ExcalidrawWrapper.tsx` — `handleChange` (drag-time wiring)

**Analog 1 — existing `onChange` interception point** (`ExcalidrawWrapper.tsx`, lines ~172–204, the exact hook D-02 attaches to):
```typescript
const handleChange = React.useCallback(
  (elements: any[], appState: any) => {
    // CRITICAL: Check suppressOnChange ref FIRST before doing anything
    if (suppressOnChangeRef?.current) {
      suppressOnChangeRef.current = false
      return // Exit early - don't call onChange, don't broadcast, don't do anything
    }

    // Call original onChange handler
    if (onChange) {
      onChange(elements, appState)
    }

    // Don't broadcast if we're receiving an update from collaborators (prevents loops)
    if (isReceivingUpdateRef?.current) {
      return
    }

    // Don't broadcast if we're loading a diagram (prevents broadcast storm)
    if (isLoadingRef?.current) {
      return
    }

    // Broadcast changes if collaborating - use the function directly from the hook
    if (isCollaborating && broadcastSceneUpdate) {
      broadcastSceneUpdate(elements, appState)
    }
  },
  [onChange, isCollaborating, broadcastSceneUpdate, isLoadingRef, isReceivingUpdateRef, suppressOnChangeRef]
)
// ...
<ExcalidrawTyped ... onChange={handleChange} ... />
```
This is the single existing guard chain (`suppressOnChangeRef` → forward to parent `onChange` → collaboration broadcast) that already fires on every drag event; D-02's new reposition logic should sit inside this function (after the `suppressOnChangeRef` early-return, so programmatic scene updates don't recursively re-trigger repositioning), operating on the `elements` array already passed in, then calling `apiRef.current.updateScene({ elements: repositionedElements }, false)` for any changed marker pairs.

**Source:** `findLinkedTextElement` (`textContainerUtils.ts` ~308) and the existing `mainElementId` field written by `createMarkerEllipses` (`sovereigntyMarkers.ts` ~206, ~220)

**Source:** `syncDiagramOnOpen` (`databaseSyncUtils.ts` ~644)
**Apply to:** `DiagramState.ts`'s scene-restore effect, per D-01 — call the exported `syncDiagramOnOpen`, do not reimplement its internals.

### Fail-soft async error handling (`try { await sync } catch { console.warn + fallback to local data }`)
**Source:** `handleOpenDiagram`'s sync block (`DiagramHandlers.ts` ~247–255)
**Apply to:** the new awaited sync call inside `DiagramState.ts`'s restore effect — never let a sync failure block scene restoration.
